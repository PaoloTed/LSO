
Indirizzi con notazione dotted

Esempio IPv4
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <arpa/inet.h> //header POSIX che fa parte dello stack di rete BSD-style
#define MYPORT 4242
int main(void) {
struct sockaddr_in my_addr;
memset(&my_addr, 0, sizeof(my_addr)); // azzera tutta la struttura
my_addr.sin_family = AF_INET; // host byte order
my_addr.sin_port = htons(MYPORT); // short →network byte order
if (inet_pton(AF_INET, "10.12.110.57", &my_addr.sin_addr) != 1) {
perror("inet_pton"); exit(1);
}
return 0;
}

return 0;

#include <netinet/in.h>

struct sockaddr_in mio_indirizzo;

mio_indirizzo.sin_family = AF_INET;
mio_indirizzo.sin_port   = htons(5200);      // network byte order
mio_indirizzo.sin_addr.s_addr = htonl(INADDR_ANY);
if (bind(fd, (struct sockaddr *)&mio_indirizzo, sizeof(mio_indirizzo)) < 0) {
perror("bind");
exit(1);
}

Send/Receive
•
accettano un parametro “flags” che permette comportamenti extra rispetto a 
write/read:
•
MSG_DONTWAIT su recv() non aspetta il dato (es. polling)
•
MSG_DONTWAIT su send() non aspetta se il buffer è pieno
•
MSG_PEEK su recv() per testare un dato prima di rimuoverlo
•
MSG_WAITALL su recv() aspetta che il len sia raggiunto
•
MSG_NOSIGNAL su send() evita il SIGPIPE
•
Senza flag stesso comportamento di read/write
ssize_t send(int sock, const void *buf, size_t len, int flags);
ssize_t recv(int sock, void *buf, size_t len, int flags);
effetto
flag
non blocca
MSG_DONTWAIT
legge senza consumare receive queue
MSG_PEEK
aspetta finché il len non è raggiunto
MSG_WAITALL
