Soluzione Esercizio n° 2 (exec)
6
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <stdlib.h>
int main(void) {
pid_t pid;
int status;
printf("Sono il programma 1 con PID %d\n", getpid());
pid = fork();
if (pid < 0) { perror("fork"); exit(1);}
if (pid == 0) {
// processo figlio
printf("Sto per eseguire prog2 con PID %d\n", getpid());
execl("./prog2", "prog2", NULL);
// se exec fallisce
perror("execl");
exit(1);
} else {
// processo padre
printf("Ho creato un figlio con PID %d\n", pid);
wait(&status);
printf("Il figlio è terminato con status %d\n", WEXITSTATUS(status));
}
return 0;
}

Soluzione Esercizio n° 3 (execlp)

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
int main(void) {
pid_t pid;
int status;
printf("[PADRE] PID = %d\n", getpid());
pid = fork();
if (pid < 0) { perror("fork"); exit(1);}
if (pid == 0) {
// Processo figlio
printf("[FIGLIO] Eseguo 'ls -l' con execlp()\n");
execlp("ls", "ls", "-l", NULL);
// Se execlp fallisce
perror("execlp");
exit(1);
} else {
// Processo padre
wait(&status);
printf("[PADRE] figlio terminato con codice %d\n", WEXITSTATUS(status));
}
return 0;
}

fflush(stdout);  // importante, perché printf è bufferizzato
// Legge da stdin (file descriptor 0)
nread = read(STDIN_FILENO, comando, sizeof(comando) - 1);

perror("read");

// Termina la stringa letta
comando[nread] = '\0';

Soluzione Esercizio n° 5 (signal)
13
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>
int counter = 0;   // contatore globale
void handler(int signo) {
counter++;
printf("\n[handler] Ricevuto SIGINT numero %d\n", counter);
if (counter >= 3) {
printf("[handler] Ho ricevuto 3 SIGINT, termino.\n");
exit(0);
}
}
int main(void) {
if (signal(SIGINT, handler) == SIG_ERR) {
perror("signal"); exit(1); }
printf("PID = %d\n", getpid());
printf("Premi Ctrl-C (3 volte per uscire)\n");
while (1) { printf("Lavoro...\n"); sleep(2); }
return 0;
}

Soluzione Esercizio n° 6 (signal)

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>
void on_sig(int s) {
if (s == SIGINT)   printf("[aspetta] Ricevuto SIGINT\n");
if (s == SIGUSR1)  printf("[aspetta] Ricevuto SIGUSR1\n");
// torna al punto interrotto
}
int main(void) {
if (signal(SIGINT,  on_sig) == SIG_ERR) { perror("signal"); exit(1); }
if (signal(SIGUSR1, on_sig) == SIG_ERR) { perror("signal"); exit(1); }
printf("PID=%d — invia SIGINT o SIGUSR1 con 'kill -SIGINT %d' / 'kill -
SIGUSR1 %d'\n",
getpid(), getpid(), getpid());
for (;;) pause();   // attende segnali indefinitamente
}
